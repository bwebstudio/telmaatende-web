import { GLSL_NOISE, snoise, surfaceHeight } from './noise'

/**
 * TELMA CONVERSATION FIELD: the engine.
 *
 * A single surface of a few thousand characters, viewed in perspective. Every
 * character is a point sprite that samples one cell of a glyph atlas drawn at
 * start-up in the site's own typeface, so nothing here is a dot dressed as a
 * letter: what reaches the screen is Suisse Intl, mipmapped.
 *
 * The surface is not simulated. Its height is a function of (x, z, time) built
 * from domain-warped simplex noise and evaluated in the vertex shader, which is
 * what makes the letters move as one sheet rather than as a swarm, and what
 * makes the whole thing cost one draw call.
 *
 * Plain WebGL2, no library. The reference (poly.ai) does the same.
 */

// --- Glyphs -----------------------------------------------------------------

// Weighted by how often they turn up in written Portuguese, so the texture of
// the field is the texture of the language rather than an even alphabet soup.
const GLYPH_WEIGHTS: [string, number][] = [
  ['a', 10], ['e', 10], ['o', 8], ['r', 6], ['s', 6], ['m', 5], ['t', 5],
  ['n', 4], ['i', 4], ['d', 3], ['u', 3], ['c', 3], ['l', 2], ['p', 2],
  ['ã', 3], ['ç', 3], ['õ', 1], ['é', 1], ['á', 1], ['ê', 1], ['ó', 1],
  ['?', 2], ['.', 3], [',', 3],
]
const ATLAS_COLS = 8
const ATLAS_ROWS = 4
const CELL = 64

// --- Composition --------------------------------------------------------------

type Preset = {
  fov: number
  eye: [number, number, number]
  target: [number, number, number]
  offset: [number, number, number]
  rotation: number
  width: number
  depth: number
  count: number
  worldSize: number
  /** How far a glyph may sit off the sheet: what turns a cloth into a volume. */
  thickness: number
  /** The travelling mass: how far its centre wanders across the field (x, z). */
  massTravel: [number, number]
  /** And its radii (x, z), before the noise tears its outline. */
  massRadius: [number, number]
  /** Where the mass's path is centred: the copy side gets visited, not lived in. */
  massBias: number
  /** Slow drift of the whole sculpture: x, y, z offset and yaw amplitudes. */
  sway: [number, number, number, number]
  dprCap: number
}

/**
 * The field is laid out far larger than the screen, and a travelling mass
 * decides which part of it is alive at any moment, so the sculpture can cross
 * the whole hero instead of deforming in place. Phones are not the desktop
 * scene scaled down: the camera comes closer and the mass wanders less, so the
 * sculpture is cropped by the screen instead of shrunk to fit it.
 */
function presetFor(width: number, lowPower: boolean): Preset {
  const scale = lowPower ? 0.6 : 1
  if (width < 768) {
    return {
      fov: 44,
      eye: [0, 2.4, 3.5],
      target: [0, 0.75, 0],
      offset: [0, 0, 0],
      rotation: -0.32,
      width: 7.0,
      depth: 5.4,
      count: Math.round(8000 * scale),
      worldSize: 0.042,
      thickness: 0.32,
      massTravel: [1.0, 0.5],
      massRadius: [1.9, 1.6],
      massBias: 0,
      sway: [0.25, 0.1, 0.25, 0.14],
      dprCap: 1.5,
    }
  }
  return {
    fov: 36,
    eye: [0, 2.7, 5.0],
    target: [0, -0.15, 0],
    offset: [0.25, 0, 0.1],
    rotation: -0.24,
    width: 12.0,
    depth: 7.0,
    count: Math.round(18000 * scale),
    worldSize: 0.044,
    thickness: 0.42,
    massTravel: [2.5, 0.8],
    massRadius: [4.3, 2.2],
    massBias: 0.3,
    sway: [0.7, 0.16, 0.6, 0.22],
    dprCap: 1.75,
  }
}

/**
 * Where the mass is at ambient time t. Two incommensurate periods per axis, so
 * the path never visibly repeats: left, through the centre, right, back, never
 * quite the same way twice. Out and back takes about forty seconds.
 */
function massCentre(p: Preset, t: number, out: Float32Array) {
  out[0] = p.massBias + p.massTravel[0] * (0.8 * Math.sin(t * 0.116 + 0.6) + 0.2 * Math.sin(t * 0.191 + 2.1))
  out[1] = p.massTravel[1] * Math.sin(t * 0.053 + 1.9)
}

/** The same envelope the shader applies. Used to decide whether a click hit. */
function presence(x: number, z: number, t: number, cx: number, cz: number, p: Preset) {
  const dx = (x - cx) / p.massRadius[0]
  const dz = (z - cz) / p.massRadius[1]
  const r = Math.hypot(dx, dz) + 0.3 * snoise(x * 0.26, z * 0.26, t * 0.045)
  const core = 1 - smoothstep(0.4, 1.1, r)
  const clumps = smoothstep(-0.5, 0.35, snoise(x * 0.5 + 9.0, z * 0.5, t * 0.07))
  const gap = smoothstep(0.0, 0.13, Math.abs(snoise(x * 0.3 - 3.0, z * 0.3, t * 0.04)))
  return core * (0.22 + 0.78 * clumps) * gap
}

// --- Shaders ---------------------------------------------------------------------

const MAX_PULSES = 6

/** How fast the ambient clock runs against the wall clock. */
const AMBIENT_SPEED = 1.35

const VERT = /* glsl */ `#version 300 es
precision highp float;

layout(location = 0) in vec3 aPos;   // model x, z, offset off the sheet
layout(location = 1) in vec3 aMeta;  // glyph, scale, density

uniform mat4 uView;
uniform mat4 uProj;
uniform vec2 uRot;          // cos, sin of the field's yaw
uniform vec3 uOffset;
uniform float uAmb;         // ambient clock (s), frozen under reduced motion
uniform float uTime;        // wall clock (s), for the pulses
uniform float uWorldSize;
uniform float uViewportH;   // device px
uniform float uMaxPoint;
uniform float uAspect;
uniform vec3 uPointer;      // ndc x, ndc y, presence 0..1
uniform vec4 uPulse[${MAX_PULSES}]; // model x, model z, start, strength
uniform vec4 uCopy;         // ndc rect the copy sits in: x0, y0, x1, y1
uniform float uIntro;
uniform vec3 uEye;
uniform vec2 uMass;         // centre of the travelling mass, model x, z
uniform vec2 uMassR;        // its radii

flat out float vGlyph;
out float vIntensity;

${GLSL_NOISE}

vec3 toWorld(vec3 m) {
  return vec3(uRot.x * m.x + uRot.y * m.z, m.y, -uRot.y * m.x + uRot.x * m.z) + uOffset;
}

// Which part of the field is alive. A ragged blob that travels, torn by black
// channels where a noise crosses zero, with clumps that gather and dissolve.
// Mirrored by presence() in the engine; keep the two in step.
float presence(vec2 p, float t) {
  vec2 d = (p - uMass) / uMassR;
  float r = length(d) + 0.3 * snoise(vec3(p * 0.26, t * 0.045));
  float core = 1.0 - smoothstep(0.4, 1.1, r);
  float clumps = smoothstep(-0.5, 0.35, snoise(vec3(p * 0.5 + vec2(9.0, 0.0), t * 0.07)));
  float gap = smoothstep(0.0, 0.13, abs(snoise(vec3(p * 0.3 - vec2(3.0, 0.0), t * 0.04))));
  return core * (0.22 + 0.78 * clumps) * gap;
}

void main() {
  vec2 p = aPos.xy;
  float t = uAmb;

  float alive = presence(p, t);
  if (alive < 0.004) {
    // Nothing to draw: skip the surface work and the fragments.
    gl_Position = vec4(2.0, 2.0, 2.0, 1.0);
    gl_PointSize = 0.0;
    vIntensity = 0.0;
    vGlyph = 0.0;
    return;
  }

  // The surface, and its slope for the light.
  const float e = 0.06;
  float h = surfaceHeight(p, t);
  float hx = surfaceHeight(p + vec2(e, 0.0), t);
  float hz = surfaceHeight(p + vec2(0.0, e), t);
  vec3 n = normalize(vec3(-(hx - h) / e, 1.0, -(hz - h) / e));

  // A drift of a few pixels, so the sheet does not only rise and fall.
  vec2 drift = 0.035 * vec2(
    snoise(vec3(p * 0.21 + 2.0, t * 0.03)),
    snoise(vec3(p * 0.21 - 4.0, t * 0.03))
  );

  // Pulses: a ring travelling out from the touch, like a pebble on cloth.
  float ripple = 0.0;
  float rippleLight = 0.0;
  for (int i = 0; i < ${MAX_PULSES}; i++) {
    vec4 pu = uPulse[i];
    float age = uTime - pu.z;
    if (pu.w <= 0.0 || age <= 0.0 || age > 7.0) continue;
    float d = distance(p, pu.xy);
    float x = d - age * 0.95;
    float width = 0.22 + age * 0.06;
    float env = exp(-(x * x) / (width * width));
    float decay = exp(-age * 0.62) * (1.0 - smoothstep(5.0, 7.0, age)) * smoothstep(0.0, 0.12, age);
    float amp = pu.w * 0.16 * decay / (1.0 + d * 0.9);
    float wave = cos(x * 9.0) * env;
    ripple += amp * wave;
    rippleLight += pu.w * decay * env * 0.8;
  }

  // Off the sheet along its normal, so the surface becomes a volume. The
  // offset breathes on its own slow phase per glyph: the vapour around the
  // sculpture is never quite still.
  float lift = aPos.z * (1.0 + 0.22 * sin(t * 0.33 + aMeta.y * 41.0));

  vec3 model = vec3(p.x + drift.x, h + ripple, p.y + drift.y) + n * lift;
  vec4 view = uView * vec4(toWorld(model), 1.0);
  vec4 clip = uProj * view;

  // The cursor. Measured on screen, so its reach is the same everywhere on the
  // sheet; it brightens and lifts the sheet by a hair, nothing more.
  vec2 ndc = clip.xy / clip.w;
  vec2 dp = (ndc - uPointer.xy) * vec2(uAspect, 1.0);
  float near = exp(-dot(dp, dp) / 0.16) * uPointer.z;
  if (near > 0.001) {
    model.y += 0.045 * near;
    view = uView * vec4(toWorld(model), 1.0);
    clip = uProj * view;
  }

  float depth = -view.z;
  float scale = aMeta.y;
  float size = uWorldSize * scale * uProj[1][1] * 0.5 * uViewportH / clip.w;
  gl_PointSize = clamp(size, 1.0, uMaxPoint);
  gl_Position = clip;
  vGlyph = aMeta.x;

  // Light: a soft key from the upper left, crests a little brighter.
  float lambert = clamp(dot(n, normalize(vec3(-0.45, 0.85, 0.5))), 0.0, 1.0);
  float lum = 0.75 + 0.6 * pow(lambert, 3.0) + 0.25 * smoothstep(-0.3, 0.45, h);
  lum += 0.55 * near + rippleLight;

  // Far letters fade into the black, so depth reads without fog.
  float depthFade = mix(0.2, 1.0, smoothstep(11.5, 4.0, depth));

  // Give the copy its air. Distance to the copy's box on screen, measured in
  // square units and torn by a slow noise, so the sculpture dissolves as it
  // nears the text rather than stopping at a rectangle.
  vec2 cc = (uCopy.xy + uCopy.zw) * 0.5;
  vec2 ch = (uCopy.zw - uCopy.xy) * 0.5;
  vec2 q = abs(ndc - cc) - ch;
  q.x *= uAspect;
  float sdf = length(max(q, 0.0)) + min(max(q.x, q.y), 0.0);
  sdf += 0.09 * snoise(vec3(ndc * 1.8, t * 0.05));
  float copyFade = mix(0.05, 1.0, smoothstep(-0.06, 0.36, sdf));

  // Where the sheet turns edge-on, its letters pile up on screen. Thin them by
  // how obliquely the sheet is seen, so a fold reads as a fold and not as a
  // bright stroke.
  vec3 worldN = vec3(uRot.x * n.x + uRot.y * n.z, n.y, -uRot.y * n.x + uRot.x * n.z);
  vec3 toEye = normalize(uEye - toWorld(model));
  float facing = abs(dot(worldN, toEye));
  float thin = mix(0.22, 1.0, smoothstep(0.08, 0.6, facing));

  float alpha = aMeta.z * pow(alive, 0.6) * depthFade * copyFade * thin * uIntro;
  vIntensity = min(lum * alpha, 1.0);
}
`

const FRAG = /* glsl */ `#version 300 es
precision mediump float;

uniform sampler2D uAtlas;
flat in float vGlyph;
in float vIntensity;
out vec4 outColor;

void main() {
  vec2 cell = vec2(mod(vGlyph, ${ATLAS_COLS}.0), floor(vGlyph / ${ATLAS_COLS}.0));
  vec2 uv = (cell + gl_PointCoord) / vec2(${ATLAS_COLS}.0, ${ATLAS_ROWS}.0);
  float ink = texture(uAtlas, uv).r;
  if (ink < 0.02) discard;
  outColor = vec4(vec3(0.93, 0.94, 0.94) * ink * vIntensity, 1.0);
}
`

// --- Small matrix helpers (column-major, no allocation per frame) -------------------

function perspective(out: Float32Array, fovDeg: number, aspect: number, near: number, far: number) {
  const f = 1 / Math.tan((fovDeg * Math.PI) / 360)
  const nf = 1 / (near - far)
  out.fill(0)
  out[0] = f / aspect
  out[5] = f
  out[10] = (far + near) * nf
  out[11] = -1
  out[14] = 2 * far * near * nf
}

type Basis = { right: number[]; up: number[]; back: number[] }

function lookAt(out: Float32Array, eye: number[], target: number[]): Basis {
  let bx = eye[0] - target[0], by = eye[1] - target[1], bz = eye[2] - target[2]
  let len = Math.hypot(bx, by, bz)
  bx /= len; by /= len; bz /= len
  // right = up(0,1,0) x back
  let rx = bz, ry = 0, rz = -bx
  len = Math.hypot(rx, ry, rz)
  rx /= len; ry /= len; rz /= len
  const ux = by * rz - bz * ry
  const uy = bz * rx - bx * rz
  const uz = bx * ry - by * rx
  out[0] = rx; out[1] = ux; out[2] = bx; out[3] = 0
  out[4] = ry; out[5] = uy; out[6] = by; out[7] = 0
  out[8] = rz; out[9] = uz; out[10] = bz; out[11] = 0
  out[12] = -(rx * eye[0] + ry * eye[1] + rz * eye[2])
  out[13] = -(ux * eye[0] + uy * eye[1] + uz * eye[2])
  out[14] = -(bx * eye[0] + by * eye[1] + bz * eye[2])
  out[15] = 1
  return { right: [rx, ry, rz], up: [ux, uy, uz], back: [bx, by, bz] }
}

// --- The field --------------------------------------------------------------------

/** Deterministic, so the sculpture is the same one on every load. */
function mulberry32(seed: number) {
  let a = seed >>> 0
  return () => {
    a = (a + 0x6d2b79f5) >>> 0
    let t = a
    t = Math.imul(t ^ (t >>> 15), t | 1)
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

function smoothstep(a: number, b: number, x: number) {
  const t = Math.min(1, Math.max(0, (x - a) / (b - a)))
  return t * t * (3 - 2 * t)
}

/**
 * The static texture of the field: where glyphs may stand at all. No outline
 * of its own any more (the travelling mass draws the silhouette), only a soft
 * fall-off at the far bounds and a fine unevenness inside, so the letters
 * gather in some places and thin out in others.
 */
function density(x: number, z: number, w: number, d: number) {
  const ex = 1 - smoothstep(0.78, 1.0, Math.abs(x) / (w / 2))
  const ez = 1 - smoothstep(0.7, 1.0, Math.abs(z) / (d / 2))
  const grain = smoothstep(
    -0.7,
    0.65,
    0.7 * snoise(x * 0.62 + 3.0, z * 0.7, 4.0) + 0.3 * snoise(x * 1.4, z * 1.4, 8.0)
  )
  return ex * ez * (0.12 + 0.73 * grain)
}

/**
 * Glyphs on a jittered grid whose rows are further apart than its columns, so
 * the rows do not collapse into one another when foreshortened. Then each one
 * is pushed off the sheet by its own amount: most stay close, a few drift far
 * out as vapour. That is what turns a sheet of letters into a cloud of them,
 * and what gives every glyph some black around it.
 */
function buildField(preset: Preset) {
  const { width: W, depth: D, count, thickness } = preset
  const rand = mulberry32(20260514)

  const glyphBag: number[] = []
  GLYPH_WEIGHTS.forEach(([, weight], i) => {
    for (let k = 0; k < weight; k++) glyphBag.push(i)
  })

  // Estimate how much of the rectangle the density keeps, then size the grid
  // so that roughly `count` glyphs survive.
  let mean = 0
  const probe = 48
  for (let i = 0; i < probe; i++) {
    for (let j = 0; j < probe; j++) {
      mean += density(((i + 0.5) / probe - 0.5) * W, ((j + 0.5) / probe - 0.5) * D, W, D)
    }
  }
  mean /= probe * probe
  const rowRatio = 1.6
  const sx = Math.sqrt((W * D * mean) / (count * rowRatio))
  const sz = sx * rowRatio

  const STRIDE = 6
  const out: number[] = []
  for (let z = -D / 2; z < D / 2; z += sz) {
    // Each row slides by its own amount, so columns never line up.
    const rowShift = rand() * sx
    for (let x = -W / 2 + rowShift; x < W / 2; x += sx * (0.85 + rand() * 0.3)) {
      const px = x + (rand() - 0.5) * sx * 0.45
      const pz = z + (rand() - 0.5) * sz * 0.5
      const dens = density(px, pz, W, D)
      if (rand() > dens) continue

      // Thicker where the noise says so: the volume is uneven too.
      const local = 0.3 + 0.7 * smoothstep(-0.35, 0.6, snoise(px * 0.38, pz * 0.38, 21.0))
      const vapour = rand() < 0.09
      const gauss = (rand() + rand() + rand() - 1.5) / 1.5
      const lift = vapour
        ? (rand() * 2 - 1) * thickness * 2.6
        : gauss * thickness * local

      const glyph = glyphBag[Math.floor(rand() * glyphBag.length)]
      const scale = 0.78 + rand() * 0.44
      const alpha = (vapour ? 0.55 : 1) * (0.45 + 0.55 * smoothstep(0.0, 0.6, dens))
      out.push(px, pz, lift, glyph, scale, alpha)
    }
  }

  // Shuffle whole records, so drawing the first N of them is an even thinning
  // of the field. That is how the frame-rate fallback reduces density.
  const n = out.length / STRIDE
  for (let i = n - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1))
    for (let k = 0; k < STRIDE; k++) {
      const tmp = out[i * STRIDE + k]
      out[i * STRIDE + k] = out[j * STRIDE + k]
      out[j * STRIDE + k] = tmp
    }
  }
  return { data: new Float32Array(out), count: n }
}

function drawAtlas(fontFamily: string) {
  const canvas = document.createElement('canvas')
  canvas.width = ATLAS_COLS * CELL
  canvas.height = ATLAS_ROWS * CELL
  const ctx = canvas.getContext('2d')!
  ctx.fillStyle = '#000'
  ctx.fillRect(0, 0, canvas.width, canvas.height)
  ctx.fillStyle = '#fff'
  ctx.textAlign = 'center'
  ctx.textBaseline = 'alphabetic'
  ctx.font = `400 46px ${fontFamily}`
  GLYPH_WEIGHTS.forEach(([ch], i) => {
    const cx = (i % ATLAS_COLS) * CELL + CELL / 2
    // One baseline for every glyph, as in set type: the comma hangs, the
    // tilde rises, and the line still reads as a line.
    const base = Math.floor(i / ATLAS_COLS) * CELL + 44
    ctx.fillText(ch, cx, base)
  })
  return canvas
}

// --- Engine ------------------------------------------------------------------------

export type FieldOptions = {
  canvas: HTMLCanvasElement
  /** Where the pointer is listened to: the hero, so the copy does not block it. */
  surface: HTMLElement
  /** Elements the sheet should thin out behind. */
  copy: () => HTMLElement | null
  fontFamily: string
}

const INTERACTIVE = 'a, button, input, select, textarea, label, summary, [role="button"]'

export function createField({ canvas, surface, copy, fontFamily }: FieldOptions) {
  const gl = canvas.getContext('webgl2', {
    alpha: false,
    antialias: false,
    depth: false,
    stencil: false,
    premultipliedAlpha: false,
    powerPreference: 'high-performance',
  })
  if (!gl) return null

  const reduced = matchMedia('(prefers-reduced-motion: reduce)')
  const finePointer = matchMedia('(hover: hover) and (pointer: fine)')
  const nav = navigator as Navigator & { deviceMemory?: number }
  const lowPower = (nav.hardwareConcurrency ?? 8) <= 4 || (nav.deviceMemory ?? 8) <= 4

  // Program.
  const compile = (type: number, src: string) => {
    const s = gl.createShader(type)!
    gl.shaderSource(s, src)
    gl.compileShader(s)
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
      throw new Error(gl.getShaderInfoLog(s) ?? 'shader compile failed')
    }
    return s
  }
  const program = gl.createProgram()!
  const vs = compile(gl.VERTEX_SHADER, VERT)
  const fs = compile(gl.FRAGMENT_SHADER, FRAG)
  gl.attachShader(program, vs)
  gl.attachShader(program, fs)
  gl.linkProgram(program)
  if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
    throw new Error(gl.getProgramInfoLog(program) ?? 'program link failed')
  }
  gl.useProgram(program)

  const u = (name: string) => gl.getUniformLocation(program, name)
  const U = {
    view: u('uView'), proj: u('uProj'), rot: u('uRot'), offset: u('uOffset'),
    amb: u('uAmb'), time: u('uTime'), worldSize: u('uWorldSize'),
    viewportH: u('uViewportH'), maxPoint: u('uMaxPoint'), aspect: u('uAspect'),
    pointer: u('uPointer'), pulse: u('uPulse'), copy: u('uCopy'),
    intro: u('uIntro'), atlas: u('uAtlas'), eye: u('uEye'),
    mass: u('uMass'), massR: u('uMassR'),
  }

  // Atlas.
  const atlas = gl.createTexture()
  gl.activeTexture(gl.TEXTURE0)
  gl.bindTexture(gl.TEXTURE_2D, atlas)
  gl.pixelStorei(gl.UNPACK_ALIGNMENT, 1)
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.R8, gl.RED, gl.UNSIGNED_BYTE, drawAtlas(fontFamily))
  gl.generateMipmap(gl.TEXTURE_2D)
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR)
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
  gl.uniform1i(U.atlas, 0)

  // Geometry.
  const vao = gl.createVertexArray()
  const vbo = gl.createBuffer()
  gl.bindVertexArray(vao)
  gl.bindBuffer(gl.ARRAY_BUFFER, vbo)
  gl.enableVertexAttribArray(0)
  gl.vertexAttribPointer(0, 3, gl.FLOAT, false, 24, 0)
  gl.enableVertexAttribArray(1)
  gl.vertexAttribPointer(1, 3, gl.FLOAT, false, 24, 12)

  gl.disable(gl.DEPTH_TEST)
  gl.enable(gl.BLEND)
  // Screen, not add: where letters overlap they brighten and then saturate
  // softly instead of clipping to a solid white bar.
  gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_COLOR)
  gl.clearColor(10 / 255, 10 / 255, 10 / 255, 1)

  const pointRange = gl.getParameter(gl.ALIASED_POINT_SIZE_RANGE) as Float32Array

  // State, all allocated once.
  const viewM = new Float32Array(16)
  const projM = new Float32Array(16)
  const pulses = new Float32Array(MAX_PULSES * 4)
  for (let i = 0; i < MAX_PULSES; i++) pulses[i * 4 + 2] = -100
  let pulseIndex = 0
  let preset = presetFor(window.innerWidth, lowPower)
  let presetKey = ''
  let basis: Basis = lookAt(viewM, preset.eye, preset.target)
  let total = 0
  let drawCount = 0
  let cssW = 1
  let cssH = 1
  let dpr = 1
  let raf = 0
  let visible = true
  let disposed = false
  let start = 0
  let last = 0
  let amb = 0
  let intro = 0
  // Where the sculpture is this frame. Shared by the shader and the picking, so
  // a click lands where the mass is now, wherever it has wandered to.
  const pose = { x: 0, y: 0, z: 0, yaw: 0 }
  const mass = new Float32Array(2)
  const updatePose = () => {
    const [sx, sy, sz, syaw] = preset.sway
    pose.x = preset.offset[0] + sx * Math.sin(amb * 0.077 + 2.4)
    pose.y = preset.offset[1] + sy * Math.sin(amb * 0.083 + 0.7)
    pose.z = preset.offset[2] + sz * Math.sin(amb * 0.039 + 4.1)
    // It turns as it travels, so it shows a different side on the way back.
    pose.yaw = preset.rotation + syaw * Math.sin(amb * 0.089 + 1.2)
    massCentre(preset, amb, mass)
  }
  const pointer = { x: 0, y: 0, tx: 0, ty: 0, amount: 0, target: 0 }
  const copyRect = new Float32Array([2, 2, 2, 2])
  let slowFrames = 0
  let sampledFrames = 0

  const rebuild = () => {
    const field = buildField(preset)
    total = field.count
    drawCount = total
    gl.bindBuffer(gl.ARRAY_BUFFER, vbo)
    gl.bufferData(gl.ARRAY_BUFFER, field.data, gl.STATIC_DRAW)
  }

  const measureCopy = () => {
    const el = copy()
    if (!el) {
      copyRect.set([2, 2, 2, 2])
      return
    }
    const c = canvas.getBoundingClientRect()
    const r = el.getBoundingClientRect()
    const toX = (px: number) => ((px - c.left) / c.width) * 2 - 1
    const toY = (py: number) => 1 - ((py - c.top) / c.height) * 2
    copyRect[0] = toX(r.left)
    copyRect[1] = toY(r.bottom)
    copyRect[2] = toX(r.right)
    copyRect[3] = toY(r.top)
  }

  const resize = () => {
    const rect = canvas.getBoundingClientRect()
    cssW = Math.max(1, rect.width)
    cssH = Math.max(1, rect.height)
    const next = presetFor(cssW, lowPower)
    const key = `${next.count}:${next.width}`
    preset = next
    if (key !== presetKey) {
      presetKey = key
      rebuild()
    }
    dpr = Math.min(window.devicePixelRatio || 1, preset.dprCap)
    canvas.width = Math.round(cssW * dpr)
    canvas.height = Math.round(cssH * dpr)
    gl.viewport(0, 0, canvas.width, canvas.height)
    perspective(projM, preset.fov, cssW / cssH, 0.1, 40)
    basis = lookAt(viewM, preset.eye, preset.target)
    measureCopy()
    if (reduced.matches) frame(performance.now())
  }

  // --- Picking: where on the sheet did the click land? ---------------------------
  const pick = (ndcX: number, ndcY: number): [number, number] | null => {
    const tanHalf = Math.tan((preset.fov * Math.PI) / 360)
    const vx = ndcX * tanHalf * (cssW / cssH)
    const vy = ndcY * tanHalf
    // World ray: right * vx + up * vy - back.
    let dx = basis.right[0] * vx + basis.up[0] * vy - basis.back[0]
    let dy = basis.right[1] * vx + basis.up[1] * vy - basis.back[1]
    let dz = basis.right[2] * vx + basis.up[2] * vy - basis.back[2]
    const len = Math.hypot(dx, dy, dz)
    dx /= len; dy /= len; dz /= len
    // Into model space: remove the offset, undo the yaw.
    const c = Math.cos(pose.yaw)
    const s = Math.sin(pose.yaw)
    const ox = preset.eye[0] - pose.x
    const oy = preset.eye[1] - pose.y
    const oz = preset.eye[2] - pose.z
    const mox = c * ox - s * oz
    const moz = s * ox + c * oz
    const mdx = c * dx - s * dz
    const mdz = s * dx + c * dz
    // March to the first sign change, then refine.
    let prevT = 0
    let prevF = oy - surfaceHeight(mox, moz, amb)
    for (let t = 0.1; t < 20; t += 0.1) {
      const f = oy + dy * t - surfaceHeight(mox + mdx * t, moz + mdz * t, amb)
      if (f <= 0) {
        let a = prevT, b = t, fa = prevF
        for (let k = 0; k < 10; k++) {
          const m = (a + b) / 2
          const fm = oy + dy * m - surfaceHeight(mox + mdx * m, moz + mdz * m, amb)
          if (Math.sign(fm) === Math.sign(fa)) { a = m; fa = fm } else b = m
        }
        const hit = (a + b) / 2
        const hx = mox + mdx * hit
        const hz = moz + mdz * hit
        // Only where there is sheet to disturb.
        const alive =
          density(hx, hz, preset.width, preset.depth) *
          presence(hx, hz, amb, mass[0], mass[1], preset)
        return alive > 0.04 ? [hx, hz] : null
      }
      prevT = t
      prevF = f
    }
    return null
  }

  const ndcFrom = (e: PointerEvent): [number, number] | null => {
    const r = canvas.getBoundingClientRect()
    const x = e.clientX - r.left
    const y = e.clientY - r.top
    if (x < 0 || y < 0 || x > r.width || y > r.height) return null
    return [(x / r.width) * 2 - 1, 1 - (y / r.height) * 2]
  }

  const emit = (ndc: [number, number]) => {
    if (reduced.matches) return
    const hit = pick(ndc[0], ndc[1])
    if (!hit) return
    const o = pulseIndex * 4
    pulses[o] = hit[0]
    pulses[o + 1] = hit[1]
    pulses[o + 2] = (performance.now() - start) / 1000
    pulses[o + 3] = 1
    pulseIndex = (pulseIndex + 1) % MAX_PULSES
  }

  // --- Input --------------------------------------------------------------------------
  let touchStart: { x: number; y: number; t: number } | null = null

  const onMove = (e: PointerEvent) => {
    if (e.pointerType === 'touch' || !finePointer.matches) return
    const ndc = ndcFrom(e)
    if (!ndc) {
      pointer.target = 0
      return
    }
    pointer.tx = ndc[0]
    pointer.ty = ndc[1]
    pointer.target = 1
  }
  const onLeave = () => {
    pointer.target = 0
  }
  const onDown = (e: PointerEvent) => {
    if ((e.target as Element | null)?.closest?.(INTERACTIVE)) return
    if (e.pointerType === 'touch') {
      // A tap, not a scroll: decided on release.
      touchStart = { x: e.clientX, y: e.clientY, t: e.timeStamp }
      return
    }
    if (e.button !== 0) return
    const ndc = ndcFrom(e)
    if (ndc) emit(ndc)
  }
  const onUp = (e: PointerEvent) => {
    if (e.pointerType !== 'touch' || !touchStart) return
    const moved = Math.hypot(e.clientX - touchStart.x, e.clientY - touchStart.y)
    const quick = e.timeStamp - touchStart.t < 400
    touchStart = null
    if (moved < 10 && quick) {
      const ndc = ndcFrom(e)
      if (ndc) emit(ndc)
    }
  }

  surface.addEventListener('pointermove', onMove, { passive: true })
  surface.addEventListener('pointerleave', onLeave)
  surface.addEventListener('pointerdown', onDown)
  surface.addEventListener('pointerup', onUp)
  const onCancel = () => {
    touchStart = null
  }
  surface.addEventListener('pointercancel', onCancel)

  // --- Frame -------------------------------------------------------------------------
  function frame(now: number) {
    if (disposed) return
    if (!start) start = now
    const dt = last ? Math.min(0.05, (now - last) / 1000) : 1 / 60
    last = now

    const still = reduced.matches
    if (still) {
      amb = 42
      intro = 1
      pointer.amount = 0
    } else {
      amb += dt * AMBIENT_SPEED
      intro = Math.min(1, intro + dt / 2.4)
      // Follows the cursor gently, and lets go of it more gently still.
      const follow = 1 - Math.exp(-dt * 5)
      pointer.x += (pointer.tx - pointer.x) * follow
      pointer.y += (pointer.ty - pointer.y) * follow
      const rate = pointer.target > pointer.amount ? 2.2 : 1.1
      pointer.amount += (pointer.target - pointer.amount) * (1 - Math.exp(-dt * rate))

      // If the device cannot hold the frame rate, thin the field rather than
      // let it stutter. Skips the first two seconds, which are noisy anyway.
      if (now - start > 2000 && drawCount > 1500) {
        sampledFrames++
        if (dt > 0.024) slowFrames++
        if (sampledFrames >= 90) {
          if (slowFrames > 45) drawCount = Math.round(drawCount * 0.75)
          sampledFrames = 0
          slowFrames = 0
        }
      }
    }

    updatePose()

    // An eased intro: the sheet surfaces out of the black.
    const introEased = intro * intro * (3 - 2 * intro)

    gl!.clear(gl!.COLOR_BUFFER_BIT)
    gl!.useProgram(program)
    gl!.bindVertexArray(vao)
    gl!.uniformMatrix4fv(U.view, false, viewM)
    gl!.uniformMatrix4fv(U.proj, false, projM)
    gl!.uniform2f(U.rot, Math.cos(pose.yaw), Math.sin(pose.yaw))
    gl!.uniform3f(U.offset, pose.x, pose.y, pose.z)
    gl!.uniform2f(U.mass, mass[0], mass[1])
    gl!.uniform2f(U.massR, preset.massRadius[0], preset.massRadius[1])
    gl!.uniform1f(U.amb, amb)
    gl!.uniform1f(U.time, (now - start) / 1000)
    gl!.uniform1f(U.worldSize, preset.worldSize)
    gl!.uniform1f(U.viewportH, canvas.height)
    gl!.uniform1f(U.maxPoint, Math.min(pointRange[1], 64 * dpr))
    gl!.uniform1f(U.aspect, cssW / cssH)
    gl!.uniform3f(U.pointer, pointer.x, pointer.y, pointer.amount)
    gl!.uniform4fv(U.pulse, pulses)
    gl!.uniform4fv(U.copy, copyRect)
    gl!.uniform1f(U.intro, introEased)
    gl!.uniform3f(U.eye, preset.eye[0], preset.eye[1], preset.eye[2])
    gl!.drawArrays(gl!.POINTS, 0, drawCount)

    if (!still && visible) raf = requestAnimationFrame(frame)
    else raf = 0
  }

  const play = () => {
    if (raf || disposed || reduced.matches || !visible || document.hidden) return
    last = 0
    raf = requestAnimationFrame(frame)
  }
  const pause = () => {
    if (raf) cancelAnimationFrame(raf)
    raf = 0
  }

  const io = new IntersectionObserver(([entry]) => {
    visible = entry.isIntersecting
    if (visible) play()
    else pause()
  })
  io.observe(canvas)

  const ro = new ResizeObserver(() => resize())
  ro.observe(canvas)
  const copyEl = copy()
  if (copyEl) ro.observe(copyEl)

  const onVisibility = () => (document.hidden ? pause() : play())
  document.addEventListener('visibilitychange', onVisibility)
  const onMotion = () => {
    if (reduced.matches) {
      pause()
      frame(performance.now())
    } else play()
  }
  reduced.addEventListener('change', onMotion)

  resize()
  if (reduced.matches) frame(performance.now())
  else play()

  return {
    dispose() {
      disposed = true
      pause()
      io.disconnect()
      ro.disconnect()
      document.removeEventListener('visibilitychange', onVisibility)
      reduced.removeEventListener('change', onMotion)
      surface.removeEventListener('pointermove', onMove)
      surface.removeEventListener('pointerleave', onLeave)
      surface.removeEventListener('pointerdown', onDown)
      surface.removeEventListener('pointerup', onUp)
      surface.removeEventListener('pointercancel', onCancel)
      gl.deleteBuffer(vbo)
      gl.deleteVertexArray(vao)
      gl.deleteTexture(atlas)
      gl.deleteProgram(program)
      gl.deleteShader(vs)
      gl.deleteShader(fs)
      gl.getExtension('WEBGL_lose_context')?.loseContext()
    },
  }
}
